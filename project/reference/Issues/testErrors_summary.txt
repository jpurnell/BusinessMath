# ✅ Drop‑In Helper

Put this in a shared test support file (e.g. `TestSupport/ErrorExpectations.swift`):

```swift
import Testing

/// Asserts that an expression throws a specific error type,
/// and gives you the thrown error for further inspection.
@discardableResult
func expectThrows<E: Error, R>(
    _ expression: @autoclosure () throws -> R,
    as errorType: E.Type = E.self,
    file: StaticString = #filePath,
    line: UInt = #line,
    _ verify: (E) -> Void = { _ in }
) rethrows -> E? {
    do {
        _ = try expression()
        Issue.record(
            "Expected error of type \(E.self), but no error was thrown.",
            file: file,
            line: line
        )
        return nil
    } catch let error as E {
        verify(error)
        return error
    } catch {
        Issue.record(
            "Expected error of type \(E.self), but got \(error).",
            file: file,
            line: line
        )
        return nil
    }
}
```
