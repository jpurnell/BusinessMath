	let e = JSONEncoder()

func json<T: Encodable>(_ value: T) throws -> String {
	String(decoding: try encoder.encode(value), as: UTF8.self)
}
/// A Swift raw-string literal for a JSON text.

func bits(_ a: [Double]) -> String { "[" + a.map { hex($0) }.joined(separator: ", ") + "]" }

// ---- selections ----
func selectionRows<N: Hashable & Comparable & Sendable & Codable>(
	name: String, graph: WeightedGraph<N>, rows: inout [String]
) async throws {
	for m in models {
		for c in configs {
			let selector = SeedSelector<N>(model: m.model, budget: c.budget, trials: c.trials, seed: c.seed)
			let selection = try selector.compute(on: graph)
			let asyncSelection = try await selector.computeAsync(on: graph)
			precondition(selection == asyncSelection, "async differs")
			let indices = selection.nodes.compactMap { graph.indexOf[$0] }
			let est = try SpreadEstimator(trials: c.trials, seed: c.seed).estimate(model: m.model, on: graph, seeds: indices)
			let envelope = try MetricEnvelope(selection)
			rows.append("\t\tGoldenSelection(graph: \"\(name)\", model: \"\(m.name)\", budget: \(c.budget), trials: \(c.trials), seed: \(c.seed), nodes: \(strs(selection.nodes.map { String(describing: $0) })), gainBits: \(bits(selection.picks.map(\.marginalGain))), totalGainBits: \(hex(selection.totalGain)), spreadMeanBits: \(hex(est.mean)), spreadStandardErrorBits: \(hex(est.standardError)), json: \(lit(try json(selection))), envelopeJSON: \(lit(try json(envelope)))),")
		}
	}
}

// ---- sweeps ----
func sweepRow(
	name: String, graphName: String, builder: String, graph: WeightedGraph<Int>,
	probabilities: [Double], budget: Int, trials: Int, seed: UInt64
) throws -> String {
	let sweep: SeedSweep<Int>
	switch builder {
	case "icw":
		sweep = try SeedSelector<Int>.sweep(
			on: graph, probabilities: probabilities, budget: budget, trials: trials, seed: seed,
			model: { try IndependentCascade(weightScaledProbability: $0) })
	case "lt":
		sweep = try SeedSelector<Int>.sweep(
			on: graph, probabilities: probabilities, budget: budget, trials: trials, seed: seed,
			model: { _ in LinearThreshold() })
	default:
		sweep = try SeedSelector<Int>.sweep(
			on: graph, probabilities: probabilities, budget: budget, trials: trials, seed: seed)
	}
	return try describe(name: name, graphName: graphName, builder: builder, budget: budget, trials: trials, seed: seed, sweep: sweep)
}

func describe(
	name: String, graphName: String, builder: String, budget: Int, trials: Int, seed: UInt64, sweep: SeedSweep<Int>
) throws -> String {
	let points = sweep.results.map {
		"GoldenSweepPoint(probabilityBits: \(hex($0.probability)), nodes: \($0.selection.nodes), gainBits: \(bits($0.selection.picks.map(\.marginalGain))))"
	}.joined(separator: ", ")
	let freq = sweep.topKFrequency.sorted { $0.key < $1.key }
	let envelope = try MetricEnvelope(sweep)
	return "\t\tGoldenSweep(name: \"\(name)\", graph: \"\(graphName)\", builder: \"\(builder)\", budget: \(budget), trials: \(trials), seed: \(seed), points: [\(points)], frequencyNodes: \(freq.map(\.key)), frequencyBits: \(bits(freq.map(\.value))), jaccardBits: [\(sweep.pairwiseJaccard.map { bits($0) }.joined(separator: ", "))], meanJaccardBits: \(hex(sweep.meanPairwiseJaccard)), alwaysPicked: \(sweep.alwaysPicked), json: \(lit(try json(sweep))), envelopeJSON: \(lit(try json(envelope)))),"
}

// ---- siting: JSON of SiteAssignment and its envelope ----
struct Inst { let name: String; let sites: Int; let demands: Int; let integer: Bool; let seed: UInt64 }
let insts: [Inst] = [
	Inst(name: "A", sites: 8, demands: 12, integer: false, seed: 11),
	Inst(name: "B", sites: 7, demands: 10, integer: true, seed: 12),
	Inst(name: "C", sites: 8, demands: 10, integer: false, seed: 13),
]
emit("\t// MARK: SiteAssignment JSON (plain and as a MetricEnvelope), for the p-median instances of the part 1 goldens")
emit("\tstatic let siteJSON: [GoldenSiteJSON] = [")
for i in insts {
	var r = SplitMix64(seed: i.seed)
	var w: [Double] = []
	var c: [[Double]] = []
	for _ in 0..<i.demands {
		w.append(i.integer ? Double(Int.random(in: 1...5, using: &r)) : Double.random(in: 0.1...10, using: &r))
		c.append((0..<i.sites).map { _ in i.integer ? Double(Int.random(in: 0...6, using: &r)) : Double.random(in: 0...20, using: &r) })
	}
	let problem = try SiteProblem(
		demand: Dictionary(uniqueKeysWithValues: w.enumerated().map { ($0, $1) }),
		sites: Array(0..<i.sites), cost: { [c] d, s in c[d][s] })
	for k in 1...3 {
		for limit in [PMedian.defaultEnumerationLimit, 0] {
			let a = try PMedian(enumerationLimit: limit).select(problem, count: k)
			let env = try MetricEnvelope(a)
			emit("\t\tGoldenSiteJSON(instance: \"\(i.name)\", count: \(k), enumerationLimit: \(limit), json: \(lit(try json(a))), envelopeJSON: \(lit(try json(env)))),")
		}
	}
}
emit("\t]")
emit("")

		emit("\t\tGoldenScoreSiting(name: \"\(sc.name)\", demandKeys: \(keys), demandBits: \(bits(keys.map { sc.scores[$0] ?? 0 })), sites: \(sc.sites), count: \(k), chosen: \(a.chosen), totalWeightBits: \(hex(problem.totalWeight)), totalCostBits: \(hex(a.totalWeightedCost)), meanCostBits: \(hex(a.meanWeightedCost)), json: \(lit(try json(a)))),")
	}
}
