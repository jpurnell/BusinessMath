// Scratch script: computes golden values with SwiftGraphMetrics (main) and writes a Swift file of constants.

func hex(_ x: Double) -> String { "0x" + String(x.bitPattern, radix: 16, uppercase: true) }
func hex(_ x: UInt64) -> String { "0x" + String(x, radix: 16, uppercase: true) }

var out = ""
func emit(_ s: String) { out += s + "\n" }

// ---- graphs ----
struct G {
	let name: String; let n: Int; let flat: DiffusionGraph
	let rng: (any DiffusionModel, [Int], inout Xoshiro256StarStar) -> Int
	init<N: Hashable & Comparable & Sendable>(name: String, graph: WeightedGraph<N>) {
		self.name = name; self.n = graph.nodeCount; self.flat = DiffusionGraph(graph)
		self.rng = { m, s, x in m.simulate(on: graph, seeds: s, using: &x) }
	}
}

func fromEdges(_ name: String, n: Int, _ edges: [(Int, Int, Double)]) throws -> G {
	G(name: name, graph: try WeightedGraph(edges: edges, isolatedNodes: Array(0..<n)))
}

// weighted12: pseudo-random but emitted as literals
var rng = SplitMix64(seed: 4242)

let weightChoices: [Double] = [0.5, 1, 1.5, 2, 2.5, 3]
for a in 0..<12 {
	for b in (a + 1)..<12 where Double.random(in: 0..<1, using: &rng) < 0.28 {
		w12.append((a, b, weightChoices[Int.random(in: 0..<6, using: &rng)]))
	}
}
let kiteG = WeightedGraph(unweightedEdges: ReferenceGraphs.kiteEdges)

// sanity: the Int-node kite built from the String one has the same adjacency as the string graph

// ---- TrialRandomness draws ----
emit("\t// MARK: TrialRandomness draws")
let pairs: [(UInt64, Int)] = [(0, 0), (1, 0), (1, 1), (2, 0), (42, 3), (2028, 17), (UInt64.max, 1_000_000), (0x9E37_79B9_7F4A_7C15, 5)]

emit("\tstatic let nodeDraws: [GoldenNodeDraw] = [")
for (s, t) in pairs {
	let r = TrialRandomness(masterSeed: s, trial: t)
	for n in nodes { emit("\t\tGoldenNodeDraw(seed: \(hex(s)), trial: \(t), node: \(n), bits: \(hex(r.uniform(forNode: n)))),") }
}
emit("\t]")
emit("\tstatic let edgeDraws: [GoldenEdgeDraw] = [")
for (s, t) in pairs {
	let r = TrialRandomness(masterSeed: s, trial: t)
	for (a, b) in edgePairs { emit("\t\tGoldenEdgeDraw(seed: \(hex(s)), trial: \(t), from: \(a), to: \(b), bits: \(hex(r.uniform(from: a, to: b)))),") }
}

// ---- models ----
struct M { let name: String; let model: any DiffusionModel }
let models: [M] = [

	M(name: "icw0.8", model: try IndependentCascade(weightScaledProbability: 0.8)),

let seedSets: [String: [[Int]]] = [
	"kite": [[0], [3, 7], [-1, 2, 2, 99]],
	"karate": [[0, 33], [5]],
	"weighted12": [[0, 5], [11]],

// ---- per-trial activated sets (derived from SGM counts by the marginal-membership trick) ----
emit("\t// MARK: Per-trial outcomes")
emit("\t// `activated` is derived from SwiftGraphMetrics alone: a node is active at the end of a trial")
emit("\t// iff it is a valid seed or adding it as an extra seed does not change the count.")
emit("\tstatic let trials: [GoldenTrial] = [")
let master: UInt64 = 7
for g in graphs {
	let flat = g.flat
	for m in models {
		for seeds in seedSets[g.name] ?? [] {
			for trial in 0..<8 {
				let r = TrialRandomness(masterSeed: master, trial: trial)
				let base = m.model.simulate(on: flat, seeds: seeds, randomness: r)
				var activated: [Int] = []
				let valid = Set(seeds.filter { $0 >= 0 && $0 < g.n })
				for v in 0..<g.n {
					if valid.contains(v) { activated.append(v); continue }
					let c = m.model.simulate(on: flat, seeds: seeds + [v], randomness: r)
					if c == base { activated.append(v) }
				}
				precondition(activated.count == base, "derivation inconsistent \(g.name) \(m.name) \(seeds) \(trial)")
				emit("\t\tGoldenTrial(graph: \"\(g.name)\", model: \"\(m.name)\", seeds: \(seeds), masterSeed: \(master), trial: \(trial), count: \(base), activated: \(activated)),")
			}
		}
	}
}
emit("\t]")
emit("")

// ---- estimates ----
emit("\t// MARK: SpreadEstimate")
emit("\tstatic let estimates: [GoldenEstimate] = [")
for g in graphs {
	for m in models {
		for seeds in seedSets[g.name] ?? [] {
			for (masterSeed, trials) in [(UInt64(1), 250), (UInt64(2028), 250), (UInt64(99), 1)] {
				let e = try SpreadEstimator(trials: trials, seed: masterSeed).estimate(model: m.model, on: g.flat, seeds: seeds)
				emit("\t\tGoldenEstimate(graph: \"\(g.name)\", model: \"\(m.name)\", seeds: \(seeds), masterSeed: \(masterSeed), trials: \(trials), meanBits: \(hex(e.mean)), standardErrorBits: \(hex(e.standardError))),")
			}
		}
	}
}
emit("\t]")
emit("")

// ---- generator overload ----
emit("\t// MARK: simulate(on:seeds:using:) with Xoshiro256StarStar")

for g in graphs {
	for m in models {
		for gseed in [UInt64(3), UInt64(17)] {
			var x = Xoshiro256StarStar(seed: gseed)
			let seeds = (seedSets[g.name] ?? [[0]])[0]
			let c = g.rng(m.model, seeds, &x)
			emit("\t\tGoldenGeneratorCount(graph: \"\(g.name)\", model: \"\(m.name)\", seeds: \(seeds), generatorSeed: \(gseed), count: \(c)),")
		}
	}
}
emit("\t]")
emit("")

// ---- siting instances ----
emit("\t// MARK: p-median instances (tables as bit patterns) and SwiftGraphMetrics's answers")
struct Inst { let name: String; let sites: Int; let demands: Int; let integer: Bool; let seed: UInt64 }
let insts: [Inst] = [
	Inst(name: "A", sites: 8, demands: 12, integer: false, seed: 11),
	Inst(name: "B", sites: 7, demands: 10, integer: true, seed: 12),
	Inst(name: "C", sites: 8, demands: 10, integer: false, seed: 13),
]
var siting = ""
emit("\tstatic let sitingInstances: [GoldenSiteInstance] = [")
var results: [String] = []
for i in insts {
	var r = SplitMix64(seed: i.seed)
	var w: [Double] = []
	var c: [[Double]] = []
	for _ in 0..<i.demands {
		w.append(i.integer ? Double(Int.random(in: 1...5, using: &r)) : Double.random(in: 0.1...10, using: &r))
		c.append((0..<i.sites).map { _ in i.integer ? Double(Int.random(in: 0...6, using: &r)) : Double.random(in: 0...20, using: &r) })
	}
	emit("\t\tGoldenSiteInstance(")
	emit("\t\t\tname: \"\(i.name)\",")
	emit("\t\t\tweightBits: [\(w.map { hex($0) }.joined(separator: ", "))],")
	emit("\t\t\tcostBits: [")
	for row in c { emit("\t\t\t\t[\(row.map { hex($0) }.joined(separator: ", "))],") }
	emit("\t\t\t]),")
	let problem = try SiteProblem(
		demand: Dictionary(uniqueKeysWithValues: w.enumerated().map { ($0, $1) }),
		sites: Array(0..<i.sites), cost: { [c] d, s in c[d][s] })
	for k in 1...3 {
		for limit in [PMedian.defaultEnumerationLimit, 0] {
			let a = try PMedian(enumerationLimit: limit).select(problem, count: k)
			results.append("\t\tGoldenSiteResult(instance: \"\(i.name)\", count: \(k), enumerationLimit: \(limit), chosen: \(a.chosen), isExact: \(a.isExact), totalBits: \(hex(a.totalWeightedCost)), meanBits: \(hex(a.meanWeightedCost)), assignedSites: \(a.assignments.map(\.site)), loadSites: \(a.loads.map(\.site)), loadBits: [\(a.loads.map { hex($0.weight) }.joined(separator: ", "))]),")
		}
	}
}
emit("\t]")
emit("\tstatic let sitingResults: [GoldenSiteResult] = [")
